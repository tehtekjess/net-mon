cmake_minimum_required(VERSION 3.20)
project(net_mon VERSION 0.5.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "net-mon requires Windows 10 or later")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(netmon_core src/core/model.cpp src/core/ip.cpp src/platform/geo.cpp src/platform/world.cpp src/platform/networks.cpp src/core/engine.cpp src/platform/windows.cpp src/storage/store.cpp src/collectors/audit.cpp src/ipc/pipe.cpp)
target_include_directories(netmon_core PUBLIC src)
target_include_directories(netmon_core SYSTEM PUBLIC third_party)
target_compile_definitions(netmon_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(netmon_core PUBLIC advapi32 wevtapi winsqlite3 ws2_32 bcrypt wintrust crypt32 shell32 ole32 psapi iphlpapi)
function(netmon_warnings target)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /sdl)
endfunction()
netmon_warnings(netmon_core)
add_executable(netmon_service src/service/main.cpp)
target_link_libraries(netmon_service PRIVATE netmon_core)
netmon_warnings(netmon_service)
add_executable(netmon WIN32 src/ui/main.cpp src/ui/table.cpp src/ui/tray.cpp src/ui/client.cpp src/ui/diagnostics.cpp src/ui/theme.cpp src/ui/map.cpp src/ui/app.rc)
target_link_libraries(netmon PRIVATE netmon_core comctl32 comdlg32 uxtheme gdiplus dwmapi)
netmon_warnings(netmon)
target_link_options(netmon PRIVATE /MANIFEST:NO)
add_executable(netmon_tests tests/tests.cpp)
target_link_libraries(netmon_tests PRIVATE netmon_core)
netmon_warnings(netmon_tests)
add_executable(netmon_ui_tests tests/table_tests.cpp src/ui/table.cpp src/ui/app.rc)
target_link_libraries(netmon_ui_tests PRIVATE netmon_core comctl32)
target_link_options(netmon_ui_tests PRIVATE /MANIFEST:NO)
netmon_warnings(netmon_ui_tests)
add_executable(netmon_responsiveness_tests tests/responsiveness_tests.cpp src/ui/table.cpp src/ui/tray.cpp src/ui/client.cpp src/ui/diagnostics.cpp src/ui/theme.cpp src/ui/map.cpp src/ui/app.rc)
target_link_libraries(netmon_responsiveness_tests PRIVATE netmon_core comctl32 comdlg32 uxtheme gdiplus dwmapi)
target_link_options(netmon_responsiveness_tests PRIVATE /MANIFEST:NO)
netmon_warnings(netmon_responsiveness_tests)

add_executable(netmon_tray_tests tests/tray_tests.cpp src/ui/tray.cpp)
target_link_libraries(netmon_tray_tests PRIVATE netmon_core)
netmon_warnings(netmon_tray_tests)

add_executable(netmon_client_tests tests/client_tests.cpp src/ui/client.cpp src/ui/table.cpp)
target_link_libraries(netmon_client_tests PRIVATE netmon_core comctl32)
netmon_warnings(netmon_client_tests)

add_executable(netmon_diagnostics_tests tests/diagnostics_tests.cpp src/ui/diagnostics.cpp)
target_link_libraries(netmon_diagnostics_tests PRIVATE netmon_core)
netmon_warnings(netmon_diagnostics_tests)

add_executable(netmon_map_tests tests/map_tests.cpp)
target_link_libraries(netmon_map_tests PRIVATE netmon_core)
netmon_warnings(netmon_map_tests)
add_executable(netmon_ui_preview tests/ui_preview.cpp src/ui/table.cpp src/ui/tray.cpp src/ui/client.cpp src/ui/diagnostics.cpp src/ui/theme.cpp src/ui/map.cpp src/ui/app.rc)
target_link_libraries(netmon_ui_preview PRIVATE netmon_core comctl32 comdlg32 uxtheme gdiplus dwmapi)
target_link_options(netmon_ui_preview PRIVATE /MANIFEST:NO)
netmon_warnings(netmon_ui_preview)

enable_testing()
add_test(NAME netmon_tests COMMAND netmon_tests)
add_test(NAME netmon_map_tests COMMAND netmon_map_tests)
add_test(NAME netmon_ui_tests COMMAND netmon_ui_tests)
add_test(NAME netmon_tray_tests COMMAND netmon_tray_tests)
add_test(NAME netmon_client_tests COMMAND netmon_client_tests)
add_test(NAME netmon_diagnostics_tests COMMAND netmon_diagnostics_tests)
install(TARGETS netmon netmon_service RUNTIME DESTINATION .)
install(FILES README.md THIRD_PARTY_NOTICES.md PLAN.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES data/geoip-country.csv data/world-map.json data/README.md DESTINATION data)
add_custom_command(TARGET netmon POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:netmon>/data" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/data/geoip-country.csv" "$<TARGET_FILE_DIR:netmon>/data/geoip-country.csv" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/data/world-map.json" "$<TARGET_FILE_DIR:netmon>/data/world-map.json")
install(DIRECTORY packaging/ DESTINATION . FILES_MATCHING PATTERN "*.ps1" PATTERN "*.cmd")
